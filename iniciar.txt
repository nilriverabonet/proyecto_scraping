# 1. Clonar tu repositorio y entrar en la carpeta
git clone https://github.com/nilriverabonet/proyecto_scraping.git
cd proyecto_scraping

# 2. Crear el entorno virtual
python -m venv .venv

# 3. Activar el entorno virtual
.\.venv\Scripts\Activate.ps1

# 4. Instalar las librerías del proyecto (requests, beautifulsoup4, pandas, streamlit)
pip install -r requirements.txt

# 5. Ejecutar el scraper y luego la web de Streamlit
python scraper.py
streamlit run app.py
